  \node[component] (login) at (4,5.7) {\textbf{LoginView / RegisterView}\\\footnotesize 注册与登录表单};
  \node[component] (profile) at (4,1.8) {\textbf{ProfileView}\\\footnotesize 资料、头像与密码维护};
  \node[component] (client) at (8.5,3.75) {\textbf{user.js + auth.js}\\\footnotesize Axios、令牌与多账号状态};

  \node[component] (security) at (13,3.75) {\textbf{SecurityConfig + JWT Filter}\\\footnotesize BCrypt 与身份认证};
  \node[component] (accountCtl) at (17.5,5.7) {\textbf{UserAccountController}\\\footnotesize 注册、登录与修改密码};

  \node[component] (mapper) at (26.5,3.75) {\textbf{UserMapper}\\\footnotesize 用户数据持久化};
  \node[datastore] (mysql) at (31,5.3) {\textbf{MySQL 8}\\\footnotesize users};
  \node[component] (avatars) at (31,1.5) {\textbf{uploads/avatars}\\\footnotesize 头像文件存储};

  \draw[request] (visitor) -- node[labeltext,above]{注册／登录} (login);
  \draw[request] (user) -- node[labeltext,above]{维护资料} (profile);

  \draw[request] (profile) -- (client);
  \draw[request] (client) -- node[labeltext,above]{HTTP + JWT} (security);
  \draw[request] (security) to[out=30,in=180] (accountCtl);
  \draw[request] (security) to[out=-30,in=180] (userCtl);
  \draw[dependency] (accountCtl) -- (accountSvc);
  \draw[dependency] (userCtl) -- (userSvc);

  \draw[dependency] (userSvc.east) to[out=-20,in=180] node[labeltext,below]{保存头像} (avatars.west);

  \groupcaption{fg}{Vue 3 前端}
  \groupcaption{bg}{Spring Boot 后端}
  \groupcaption{dg}{数据与文件基础设施}
